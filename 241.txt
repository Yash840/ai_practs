import heapq

goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)

# Calculate Manhattan Distance
def heuristic(state):
    h = 0

    for i, n in enumerate(state):
        if n != 0:
            j = goal.index(n)
            h += abs(i//3 - j//3) + abs(i%3 - j%3)

    return h

# A* Search Algorithm
def astar(start):
    pq = [(heuristic(start), 0, start)]
    parent = {start: None}
    cost = {start: 0}

    while pq:
        f, g, state = heapq.heappop(pq)

        if g != cost[state]:
            continue

        if state == goal:
            path = []

            while state is not None:
                path.append(state)
                state = parent[state]

            return path[::-1]

        z = state.index(0)
        r, c = z//3, z%3

        moves = [(0,1), (0,-1), (1,0), (-1,0)]

        for dr, dc in moves:
            nr, nc = r+dr, c+dc

            if 0 <= nr < 3 and 0 <= nc < 3:
                nz = nr*3 + nc

                new = list(state)
                new[z], new[nz] = new[nz], new[z]
                new = tuple(new)

                new_cost = g + 1

                if new_cost < cost.get(new, float('inf')):
                    cost[new] = new_cost
                    parent[new] = state

                    heapq.heappush(
                        pq, (new_cost + heuristic(new),
                             new_cost, new)
                    )

    return None

# Initial State

from collections import deque

def hanoi(n):
    start = (0,) * n
    goal = (2,) * n

    queue = deque([start])
    visited = {start: None}

    while queue:
        state = queue.popleft()

        if state == goal:
            path = []

            while state is not None:
                path.append(state)
                state = visited[state]

            return path[::-1]

        for disk in range(n):
            source = state[disk]

            # Check if disk is on top
            if any(state[i] == source for i in range(disk)):
                continue

            for dest in range(3):
                if dest == source:
                    continue

                # Check legal movement
                if any(state[i] == dest for i in range(disk)):
                    continue

                new = list(state)
                new[disk] = dest
                new = tuple(new)

                if new not in visited:
                    visited[new] = state
                    queue.append(new)

    return []

# Solve for 3 disks
path = hanoi(3)

for state in path:
    print(state)

    for i in range(3):
        for j in range(3):

A4. ALPHA BETA PRUNING
def minimax(depth, node, maximizing, alpha, beta, values):

    if depth == 3:
        return values[node]

    if maximizing:
        best = float('-inf')

        for i in range(2):
            val = minimax(depth+1, node*2+i,
                          False, alpha, beta, values)

            best = max(best, val)
            alpha = max(alpha, best)

            if beta <= alpha:
                break

        return best

    else:
        best = float('inf')

        for i in range(2):
            val = minimax(depth+1, node*2+i,
                          True, alpha, beta, values)

            best = min(best, val)
            beta = min(beta, best)

            if beta <= alpha:
                break

        return best


values = [3, 5, 6, 9, 1, 2, 0, -1]

result = minimax(0, 0, True,
                 float('-inf'), float('inf'), values)

print("Optimal value:", result)

import random

cities = 5
k = 2

distance = [
    [0, 2, 9, 10, 7],
    [2, 0, 6, 4, 3],
    [9, 6, 0, 8, 5],
    [10, 4, 8, 0, 6],
    [7, 3, 5, 6, 0]
]

def cost(route):
    total = 0
    for i in range(cities):
        total += distance[route[i]][route[(i+1) % cities]]
    return total

# Initial beam
beam = [random.sample(range(cities), cities) for _ in range(k)]

for _ in range(100):
    new_beam = []

    for route in beam:
        for i in range(cities):
            for j in range(i+1, cities):
                new = route[:]
                new[i], new[j] = new[j], new[i]
                new_beam.append(new)

    beam = sorted(new_beam, key=cost)[:k]

best = min(beam, key=cost)

print("Best Route:", best + [best[0]])
print("Minimum Distance:", cost(best))

rules = {
    "Flu": ["fever", "cough", "body_pain"],
    "Cold": ["sneezing", "runny_nose", "cough"],
    "Allergy": ["sneezing", "itchy_eyes"],
    "Malaria": ["fever", "chills", "sweating"]
}

facts = {"fever", "cough", "body_pain"}

# Forward Chaining
print("Forward Chaining:")

for disease, symptoms in rules.items():
    if all(s in facts for s in symptoms):
        facts.add(disease)
        print("Possible condition:", disease)

# Backward Chaining
def backward(goal):
    if goal in facts:
        return True

    if goal not in rules:
        return False

    return all(backward(s) for s in rules[goal])

print("\nBackward Chaining:")

goal = "Flu"

if backward(goal):
    print("Goal proved:", goal)
else:
    print("Goal not proved:", goal)
